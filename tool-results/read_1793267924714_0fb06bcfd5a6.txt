    96→// --- Page Views ---
    97→type PageView = "landing" | "login" | "signup" | "app";
    98→
    99→export default function Home() {
   100→  const user = useAuthStore((s) => s.user);
   101→  const authLoaded = useAuthStore((s) => s.isLoaded);
   102→  const configLoaded = useConfigStore((s) => s.isLoaded);
   103→
   104→  // Page view state — starts at landing, changes via user actions or auto-login
   105→  const [view, setView] = useState<PageView>("landing");
   106→
   107→  // Success message displayed on the login screen after a successful signup
   108→  const [signupSuccess, setSignupSuccess] = useState<string | null>(null);
   109→
   110→  // Auto-redirect to app if already logged in (from localStorage hydration).
   111→  // On redirige dès qu'un user est présent en localStorage, peu importe
   112→  // isOnboarded. La vue "app" gère elle-même l'onboarding si isOnboarded=false
   113→  // (elle affiche alors OnboardingWizard). Cela évite que LoginOrSignupView
   114→  // ne se mette à afficher l'onboarding à la place du formulaire de login/signup.
   115→  const hasAutoRedirected = useState(false);
   116→  if (
   117→    !hasAutoRedirected[0] &&
   118→    authLoaded &&
   119→    configLoaded &&
   120→    user &&
   121→    view === "landing"
   122→  ) {
   123→    hasAutoRedirected[1](true);
   124→    setView("app");
   125→  }
   126→
   127→  // CRITICAL: When user becomes null (logout), immediately switch back to landing
   128→  // with NO transition delay. This ensures the app view unounts instantly.
   129→  // We compute the effective view synchronously (during render) AND mirror it
   130→  // into state via useEffect so the change persists across re-renders.
   131→  useEffect(() => {
   132→    if (!user && view === "app") {
   133→      setView("landing");
   134→    }
   135→  }, [user, view]);
   136→
   137→  // Si l'utilisateur est déjà connecté (via login ou hydratation localStorage)
   138→  // mais que view est encore "login" ou "signup", on bascule en "app".
   139→  // Cela évite qu'un user connecté reste bloqué sur la page de login/signup.
   140→  useEffect(() => {
   141→    if (user && (view === "login" || view === "signup")) {
   142→      setView("app");
   143→    }
   144→  }, [user, view]);
   145→
   146→  // Synchronous fallback — instant view switch with no intermediate render
   147→  // that could show a "Redirection..." placeholder.
   148→  const effectiveView: PageView =
   149→    authLoaded && configLoaded && !user && view === "app" ? "landing" : view;
   150→
   151→  const handleLoginSuccess = () => {
   152→    setView("app");
   153→  };
   154→
   155→  const handleGoToLogin = () => {
   156→    setSignupSuccess(null);
   157→    setView("login");
   158→  };
   159→
   160→  const handleGoToSignup = () => {
   161→    setSignupSuccess(null);
   162→    setView("signup");
   163→  };
   164→
   165→  const handleBackToHome = () => {
   166→    setSignupSuccess(null);
   167→    setView("landing");
   168→  };
   169→
   170→  // Immediate logout: clear auth state + switch view synchronously.
   171→  // No setTimeout, no async — avoids the 300ms exit transition delay from AnimatePresence.
   172→  const handleLogout = () => {
   173→    useAuthStore.getState().logout();
   174→    setView("landing");
   175→  };
   176→
   177→  // Loading
   178→  if (!authLoaded || !configLoaded) {
   179→    return <LoadingScreen />;
   180→  }
   181→
   182→  return (
   183→    <AnimatePresence mode="wait">
   184→      {effectiveView === "landing" && (
   185→        <motion.div
   186→          key="landing"
   187→          initial={{ opacity: 0 }}
   188→          animate={{ opacity: 1 }}
   189→          exit={{ opacity: 0 }}
   190→          transition={{ duration: 0.2 }}
   191→          className="min-h-screen flex flex-col bg-background"
   192→        >
   193→          <Navbar
   194→            onLogin={handleGoToLogin}
   195→            onSignup={handleGoToSignup}
   196→          />
   197→          <main className="flex-1">
   198→            <Hero onSignup={handleGoToSignup} />
   199→            <Features onSignup={handleGoToSignup} />
   200→            <Benefits onSignup={handleGoToSignup} />
   201→            <HowItWorks onSignup={handleGoToSignup} />
   202→            <Statistics />
   203→            <MoreBenefits onSignup={handleGoToSignup} />
   204→            <GlobalTransfer onSignup={handleGoToSignup} />
   205→            <Referral onSignup={handleGoToSignup} />
   206→            <CompleteSolutions onSignup={handleGoToSignup} />
   207→            <Testimonials />
   208→            <BigStat />
   209→            <Pricing onSignup={handleGoToSignup} />
   210→            <FAQ />
   211→            <FinalCTA onSignup={handleGoToSignup} />
   212→          </main>
   213→          <Footer />
   214→          <ScrollToTop />
   215→        </motion.div>
   216→      )}
   217→
   218→      {(effectiveView === "login" || effectiveView === "signup") && (
   219→        <motion.div
   220→          key={effectiveView}
   221→          initial={{ opacity: 0, y: 20 }}
   222→          animate={{ opacity: 1, y: 0 }}
   223→          exit={{ opacity: 0, y: -20 }}
   224→          transition={{ duration: 0.2 }}
   225→        >
   226→          <LoginOrSignupView
   227→            mode={effectiveView as "login" | "signup"}
   228→            onLoginSuccess={handleLoginSuccess}
   229→            onBackToHome={handleBackToHome}
   230→            signupSuccess={signupSuccess}
   231→            onSignupSuccess={(message) => {
   232→              setSignupSuccess(message || null);
   233→              setView("login");
   234→            }}
   235→          />
   236→        </motion.div>
   237→      )}
   238→
   239→      {effectiveView === "app" && (
   240→        <motion.div
   241→          key="app"
   242→          initial={{ opacity: 0 }}
   243→          animate={{ opacity: 1 }}
   244→          // No exit transition — when user logs out, the view is removed instantly
   245→          transition={{ duration: 0.2 }}
   246→        >
   247→          <AppView
   248→            onLogout={handleLogout}
   249→            onBackToHome={handleBackToHome}
   250→          />
   251→        </motion.div>
   252→      )}
   253→    </AnimatePresence>
   254→  );
   255→}

   257→// --- Login / Signup View ---
   258→function LoginOrSignupView({
   259→  mode,
   260→  onLoginSuccess,
   261→  onBackToHome,
   262→  signupSuccess,
   263→  onSignupSuccess,
   264→}: {
   265→  mode: "login" | "signup";
   266→  onLoginSuccess: () => void;
   267→  onBackToHome: () => void;
   268→  signupSuccess?: string | null;
   269→  onSignupSuccess: (message: string) => void;
   270→}) {
   271→  // IMPORTANT : LoginOrSignupView ne doit JAMAIS afficher OnboardingWizard.
   272→  // Si l'utilisateur arrive ici, c'est qu'il n'est PAS encore connecté
   273→  // (sinon l'auto-redirect l'aurait envoyé en vue "app" qui gère elle-même
   274→  // l'onboarding). On affiche donc simplement SignupScreen ou LoginScreen
   275→  // selon le mode demandé.
   276→
   277→  if (mode === "signup") {
   278→    return (
   279→      <SignupScreen
   280→        onLoginSuccess={onLoginSuccess}
   281→        onBackToHome={onBackToHome}
   282→        onGotoLogin={onSignupSuccess}
   283→      />
   284→    );
   285→  }
   286→
   287→  return (
   288→    <div className="min-h-screen bg-background">
   289→      <div className="absolute top-4 left-4 z-50">
   290→        <Button
   291→          variant="ghost"
   292→          size="sm"
   293→          className="text-foreground/60 hover:text-foreground gap-1"
   294→          onClick={onBackToHome}
   295→        >
   296→          ← Retour au site
   297→        </Button>
   298→      </div>
   299→      <LoginScreen
   300→        onLoginSuccess={onLoginSuccess}
   301→        successMessage={signupSuccess ?? null}
   302→      />
   303→    </div>
   304→  );
   305→}

   307→// --- Signup Screen ---
   308→function SignupScreen({
   309→  onLoginSuccess: _onLoginSuccess,
   310→  onBackToHome,
   311→  onGotoLogin,
   312→}: {
   313→  onLoginSuccess: () => void;
   314→  onBackToHome: () => void;
   315→  onGotoLogin: (successMessage: string) => void;
   316→}) {
   317→  const [step, setStep] = useState<1 | 2>(1);
   318→
   319→  // Step 1 fields
   320→  const [name, setName] = useState("");
   321→  const [email, setEmail] = useState("");
   322→  const [phone, setPhone] = useState("");
   323→  const [businessName, setBusinessName] = useState("");
   324→
   325→  // Step 2 fields
   326→  const [identifier, setIdentifier] = useState("");
   327→  const [password, setPassword] = useState("");
   328→  const [confirmPassword, setConfirmPassword] = useState("");
   329→  const [showPassword, setShowPassword] = useState(false);
   330→  const [loading, setLoading] = useState(false);
   331→  const [error, setError] = useState<string | null>(null);
   332→
   333→  const canSubmitStep1 = Boolean(name && email && businessName);
   334→  const passwordsMatch = password.length > 0 && password === confirmPassword;
   335→  const canSubmitStep2 =
   336→    Boolean(identifier) && password.length >= 6 && passwordsMatch;
   337→
   338→  const handleStep1Submit = (e: React.FormEvent) => {
   339→    e.preventDefault();
   340→    if (!canSubmitStep1) return;
   341→    setError(null);
   342→    setStep(2);
   343→  };
   344→
   345→  const handleStep2Submit = async (e: React.FormEvent) => {
   346→    e.preventDefault();
   347→    if (!canSubmitStep2) {
   348→      if (password.length > 0 && password !== confirmPassword) {
   349→        setError("Les mots de passe ne correspondent pas.");
   350→      } else if (password.length > 0 && password.length < 6) {
   351→        setError("Le mot de passe doit contenir au moins 6 caractères.");
   352→      } else {
   353→        setError("Veuillez remplir tous les champs.");
   354→      }
   355→      return;
   356→    }
   357→
   358→    setLoading(true);
   359→    setError(null);
   360→
   361→    try {
   362→      const res = await fetch("/api/auth/register", {
   363→        method: "POST",
   364→        headers: { "Content-Type": "application/json" },
   365→        body: JSON.stringify({
   366→          name,
   367→          email,
   368→          phone,
   369→          businessName,
   370→          identifier,
   371→          password,
   372→        }),
   373→      });
   374→
   375→      const data = await res.json().catch(() => ({}));
   376→
   377→      if (res.ok) {
   378→        // Redirect to login with success message — NO auto-login
   379→        onGotoLogin("Compte créé ! Connectez-vous.");
   380→      } else {
   381→        setError(data.error ?? "Échec de la création du compte");
   382→      }
   383→    } catch {
   384→      setError("Impossible de contacter le serveur. Vérifiez votre connexion.");
   385→    } finally {
   386→      setLoading(false);
   387→    }
   388→  };
   389→
   390→  return (
   391→    <div className="min-h-screen flex items-center justify-center bg-background p-4">
   392→      <motion.div
   393→        initial={{ opacity: 0, y: 20 }}
   394→        animate={{ opacity: 1, y: 0 }}
   395→        className="w-full max-w-md"
   396→      >
   397→        {/* Back button */}
   398→        <Button
   399→          variant="ghost"
   400→          size="sm"
   401→          className="mb-6 text-foreground/60 hover:text-foreground gap-1"
   402→          onClick={() => (step === 2 ? setStep(1) : onBackToHome())}
   403→        >
   404→          ← {step === 2 ? "Retour" : "Retour"}
   405→        </Button>
   406→
   407→        {/* Header */}
   408→        <div className="text-center mb-8">
   409→          <div className="w-16 h-16 rounded-2xl bg-yk-deep flex items-center justify-center mx-auto mb-4 shadow-lg">
   410→            <span className="text-white font-bold text-2xl">YK</span>
   411→          </div>
   412→          <h1 className="text-2xl font-bold text-yk-deep">
   413→            {step === 1 ? "Créer votre compte" : "Sécuriser votre compte"}
   414→          </h1>
   415→          <p className="text-sm text-muted-foreground mt-1">
   416→            {step === 1
   417→              ? "14 jours d'essai gratuit — aucune carte requise"
   418→              : "Choisissez un identifiant et un mot de passe"}
   419→          </p>
   420→        </div>
   421→
   422→        {/* Step indicator */}
   423→        <div className="flex items-center justify-center gap-2 mb-6">
   424→          <div className={`h-1.5 w-12 rounded-full ${step === 1 ? "bg-yk-deep" : "bg-yk-deep/40"}`} />
   425→          <div className={`h-1.5 w-12 rounded-full ${step === 2 ? "bg-yk-deep" : "bg-yk-deep/20"}`} />
   426→        </div>
   427→
   428→        <Card>
   429→          <CardContent className="p-6">
   430→            {step === 1 ? (
   431→              <form onSubmit={handleStep1Submit} className="space-y-4">
   432→                <div className="space-y-2">
   433→                  <label className="text-xs font-medium" htmlFor="signup-name">Nom complet</label>
   434→                  <input
   435→                    id="signup-name"
   436→                    type="text"
   437→                    placeholder="Ibrahima Fall"
   438→                    value={name}
   439→                    onChange={(e) => setName(e.target.value)}
   440→                    className="w-full h-9 rounded-md border border-input bg-transparent px-3 py-1 text-sm shadow-sm focus:outline-none focus:ring-1 focus:ring-ring"
   441→                    required
   442→                  />
   443→                </div>
   444→
   445→                <div className="space-y-2">
   446→                  <label className="text-xs font-medium" htmlFor="signup-email">Email</label>
   447→                  <input
   448→                    id="signup-email"
   449→                    type="email"
   450→                    placeholder="ibrahima@yonukom.sn"
   451→                    value={email}
   452→                    onChange={(e) => setEmail(e.target.value)}
   453→                    className="w-full h-9 rounded-md border border-input bg-transparent px-3 py-1 text-sm shadow-sm focus:outline-none focus:ring-1 focus:ring-ring"
   454→                    required
   455→                  />
   456→                </div>
   457→
   458→                <div className="space-y-2">
   459→                  <label className="text-xs font-medium" htmlFor="signup-phone">Téléphone (optionnel)</label>
   460→                  <input
   461→                    id="signup-phone"
   462→                    type="tel"
   463→                    placeholder="+221 77 123 45 67"
   464→                    value={phone}
   465→                    onChange={(e) => setPhone(e.target.value)}
   466→                    className="w-full h-9 rounded-md border border-input bg-transparent px-3 py-1 text-sm shadow-sm focus:outline-none focus:ring-1 focus:ring-ring"
   467→                  />
   468→                </div>
   469→
   470→                <div className="space-y-2">
   471→                  <label className="text-xs font-medium" htmlFor="signup-biz">Nom du commerce</label>
   472→                  <input
   473→                    id="signup-biz"
   474→                    type="text"
   475→                    placeholder="Boutique YonuKom"
   476→                    value={businessName}
   477→                    onChange={(e) => setBusinessName(e.target.value)}
   478→                    className="w-full h-9 rounded-md border border-input bg-transparent px-3 py-1 text-sm shadow-sm focus:outline-none focus:ring-1 focus:ring-ring"
   479→                    required
   480→                  />
   481→                </div>
   482→
   483→                <Button
   484→                  type="submit"
   485→                  className="w-full bg-yk-deep hover:bg-yk-deep-light text-white font-medium gap-2"
   486→                  disabled={!canSubmitStep1}
   487→                >

   490→                </Button>
   491→              </form>
   492→            ) : (
   493→              <form onSubmit={handleStep2Submit} className="space-y-4">
   494→                <div className="space-y-2">
   495→                  <label className="text-xs font-medium" htmlFor="signup-identifier">Identifiant</label>
   496→                  <input
   497→                    id="signup-identifier"
   498→                    type="text"
   499→                    placeholder="ibrahima.fall"
   500→                    value={identifier}
   501→                    onChange={(e) => setIdentifier(e.target.value)}
   502→                    className="w-full h-9 rounded-md border border-input bg-transparent px-3 py-1 text-sm shadow-sm focus:outline-none focus:ring-1 focus:ring-ring"
   503→                    required
   504→                    autoComplete="username"
   505→                  />
   506→                  <p className="text-[10px] text-muted-foreground">
   507→                    Utilisé pour vous connecter avec votre email.

   811→  orderCount: number;
   812→  avgOrderValue: number;
   813→  totalStock: number;

   816→function useTodayKpis(businessId?: string) {
   817→  const [data, setData] = useState<KpiData>({
   818→    totalRevenue: 0,
   819→    orderCount: 0,
   820→    avgOrderValue: 0,
   821→    totalStock: 0,
   822→  });
   823→  const [loading, setLoading] = useState<boolean>(true);
   824→  const [error, setError] = useState<boolean>(false);
   825→
   826→  useEffect(() => {
   827→    let cancelled = false;
   828→    if (!businessId || businessId === "demo") {
   829→      setData({ totalRevenue: 0, orderCount: 0, avgOrderValue: 0, totalStock: 0 });
   830→      setLoading(false);
   831→      return;
   832→    }
   833→    setLoading(true);
   834→    Promise.all([
   835→      fetch(`/api/analytics?businessId=${encodeURIComponent(businessId)}&period=today`).then((r) => r.ok ? r.json() : null),
   836→      fetch(`/api/products?businessId=${encodeURIComponent(businessId)}`).then((r) => r.ok ? r.json() : null),
   837→    ])
   838→      .then(([a, p]) => {
   839→        if (cancelled) return;
   840→        const m = a?.metrics ?? {};
   841→        const productsList: Array<{ quantity?: number; stock?: number }> = p?.products ?? p ?? [];
   842→        const totalStock = productsList.reduce(
   843→          (sum: number, pr: { quantity?: number; stock?: number }) =>
   844→            sum + (typeof pr.quantity === "number" ? pr.quantity : typeof pr.stock === "number" ? pr.stock : 0),
   845→          0,
   846→        );
   847→        setData({
   848→          totalRevenue: m.totalRevenue ?? 0,
   849→          orderCount: m.orderCount ?? 0,
   850→          avgOrderValue: m.avgOrderValue ?? 0,
   851→          totalStock,
   852→        });
   853→        setError(false);
   854→      })
   855→      .catch(() => {
   856→        if (cancelled) return;
   857→        setError(true);
   858→      })
   859→      .finally(() => {
   860→        if (!cancelled) setLoading(false);
   861→      });
   862→    return () => {
   863→      cancelled = true;
   864→    };
   865→  }, [businessId]);
   866→
   867→  return { data, loading, error };
   868→}
