   109→/// Produit avec distinction CRITIQUE entre stock physique et stock réservé.
   110→/// 
   111→/// ⚠️ LOGIQUE MÉTIER IMPORTANTE :
   112→/// - quantity = Stock PHYSIQUE (ce qui est réellement en magasin)
   113→/// - reservedQuantity = Stock RÉSERVÉ (commandes validées en devis mais pas encore livrées)
   114→/// - Stock DISPONIBLE = quantity - reservedQuantity
   115→/// - Un devis validé DÉDUIT du stock disponible mais PAS du stock physique
   116→/// - Une livraison DÉDUIT du stock physique et libère la réservation
   117→model Product {
   118→  id                String   @id @default(cuid())
   119→  name              String   // Nom du produit (ex: "Thiébou Rouge", "Sac riz 25kg")
   120→  sku               String?  // Code produit / référence
   121→  description       String?
   122→  /// Catégorie (texte) — DEPRECATED, utiliser categoryId (FK vers Category)
   123→  /// Conservé pour la rétro-compatibilité avec les produits existants.
   124→  category          String?
   125→  /// Lien vers une Category créée par le commerçant (optionnel)
   126→  categoryId        String?
   127→  categoryRef       Category? @relation(fields: [categoryId], references: [id])
   128→  imageUrl          String?
   129→
   130→  // --- PRIX ---
   131→  /// Prix d'achat en FCFA — ⚠️ NON VISIBLE par le vendeur (règle anti-fraude)
   132→  purchasePrice     Int      @default(0)
   133→  /// Prix de vente en FCFA
   134→  salePrice         Int
   135→
   136→  // --- STOCK (DISTINCTION CRITIQUE) ---
   137→  /// Stock physique total en magasin
   138→  quantity          Int      @default(0)
   139→  /// Stock réservé par des devis B2B validés (pas encore livrés)
   140→  reservedQuantity  Int      @default(0)
   141→  /// Seuil d'alerte stock bas — notification quand quantity <= minStockAlert
   142→  minStockAlert     Int      @default(5)
   143→
   144→  // --- TYPE DE PRODUIT ---
   145→  /// "standard" = produit normal
   146→  /// "plate_restaurant" = plat fini (restaurant) — vente déduit 1 unité
   147→  /// "service" = prestation (pas de stock déduit)
   148→  productType       String   @default("standard")
   149→  /// "piece" | "kg" | "litre" | "plat"
   150→  unitType          String   @default("piece")
   151→
   152→  isActive          Boolean  @default(true)
   153→
   154→  // --- BOUTIQUE EN LIGNE (V2) ---
   155→  /// Produit vedette — mis en avant en haut de la boutique publique
   156→  isFeatured        Boolean  @default(false)
   157→  /// Prix barré (si renseigné et > salePrice → badge "promo" + ancien prix affiché barré)
   158→  compareAtPrice    Int?
   159→
   160→  businessId        String
   161→  business          Business @relation(fields: [businessId], references: [id])
   162→
   163→  orderItems        OrderItem[]
   164→  stockMovements    StockMovement[]
   165→  purchaseItems     PurchaseItem[]
   166→  reviews           Review[]
   167→
   168→  createdAt         DateTime @default(now())
   169→  updatedAt         DateTime @updatedAt
   170→
   171→  @@index([categoryId])
   172→  @@index([isFeatured])
   173→}

   175→/// Catégorie de produit créée par le commerçant (remplace le select hardcoded).
   176→/// Chaque catégorie est propre à un Business — modèle multi-tenant simple.
   177→model Category {
   178→  id          String   @id @default(cuid())
   179→  businessId  String
   180→  business    Business @relation(fields: [businessId], references: [id])
   181→  /// Nom lisible : "Boissons", "Plats", "Cosmétiques", etc.
   182→  name        String
   183→  /// Couleur optionnelle (hex) pour différencier visuellement les chips
   184→  color       String?
   185→  /// Image optionnelle (URL servie par /api/uploads) affichée sur la carte
   186→  /// de catégorie et dans la caisse. Permet une identité visuelle forte
   187→  /// (ex: photo du plat, logo de la gamme).
   188→  imageUrl    String?
   189→  /// Ordre d'affichage (les plus petits en premier)
   190→  sortOrder   Int      @default(0)
   191→  /// Une catégorie peut être désactivée sans être supprimée
   192→  isActive    Boolean  @default(true)
   193→
   194→  products    Product[]
   195→  createdAt   DateTime @default(now())
   196→  updatedAt   DateTime @updatedAt
   197→
   198→  @@index([businessId])
   199→  @@unique([businessId, name])
   200→}

   202→/// Journal de tous les mouvements de stock — traçabilité complète.
   203→/// Chaque vente, restock, réservation, perte génère un mouvement.
   204→model StockMovement {
   205→  id              String   @id @default(cuid())
   206→
   207→  /// "sale" = vente (déduit stock physique)
   208→  /// "restock" = réapprovisionnement
   209→  /// "reservation" = devis B2B validé (augmente reservedQuantity)
   210→  /// "reservation_release" = livraison B2B (libère réservation + déduit physique)
   211→  /// "loss" = perte/gaspillage (déduit stock physique SANS impacter le CA)
   212→  /// "adjustment" = ajustement manuel (nécessite un motif)
   213→  type            String
   214→
   215→  /// Quantité du mouvement (toujours positif — le type détermine le sens)
   216→  quantity        Int
   217→
   218→  /// Motif obligatoire pour "loss" et "adjustment"
   219→  reason          String?
   220→
   221→  /// Flag spécifique aux pertes (restaurant) :
   222→  /// Une perte déduit le stock mais n'apparaît PAS dans le Chiffre d'Affaires
   223→  isLoss          Boolean  @default(false)
   224→
   225→  productId       String
   226→  product         Product  @relation(fields: [productId], references: [id])
   227→
   228→  /// Référence à la commande si le mouvement est lié à une vente
   229→  orderId         String?
   230→  /// Utilisateur qui a effectué le mouvement
   231→  userId          String?
   232→
   233→  /// Marqueur offline — ce mouvement a été créé hors ligne
   234→  isOfflineCreated Boolean @default(false)
   235→  syncedAt        DateTime?
   236→
   237→  createdAt       DateTime @default(now())
   238→}

   280→/// Commande avec statuts variables selon le module actif.
   281→///
   282→/// Module E-Commerce Social : "to_pay" → "to_ship" → "shipped" | "cancelled"
   283→/// Module Pipeline B2B : "draft" → "quote_validated" → "delivery_note" → "invoiced" | "cancelled"
   284→/// Module Restaurant : "pending" → "completed" | "cancelled"
   285→model Order {
   286→  id              String   @id @default(cuid())
   287→  /// Numéro lisible et unique (ex: YK-2025-0001)
   288→  orderNumber     String   @unique
   289→
   290→  /// Statut de la commande (dépend du module — voir commentaires ci-dessus)
   291→  status          String   @default("draft")
   292→
   293→  // --- MONTANTS EN FCFA (entiers — JAMAIS de flottants avec la monnaie) ---
   294→  /// Somme des OrderItem.totalPrice avant remises
   295→  subtotal        Int      @default(0)
   296→  /// Montant total TTC
   297→  totalAmount     Int      @default(0)
   298→  /// Total des paiements reçus (somme de Payment.amount)
   299→  paidAmount      Int      @default(0)
   300→  /// Reste à payer = totalAmount - paidAmount (calculé automatiquement)
   301→  remainingAmount Int      @default(0)
   302→
   303→  // --- LIVRAISON ---
   304→  deliveryAddress String?
   305→  deliveryDate    DateTime?
   306→  isDelivered     Boolean  @default(false)
   307→  /// Frais de livraison (FCFA) — 0 si pas de livraison
   308→  deliveryFee     Int      @default(0)
   309→  /// Type de commande: "dine_in" | "takeaway" | "delivery"
   310→  orderType       String   @default("dine_in")
   311→
   312→  // --- E-COMMERCE SOCIAL ---
   313→  /// WhatsApp de confirmation envoyé ?
   314→  whatsappSent    Boolean  @default(false)
   315→  /// Lien de paiement Wave/OM collé par le client
   316→  paymentLink     String?
   317→
   318→  // --- ANTI-FRADE : ANNULATION CONTRÔLÉE ---
   319→  /// Un vendeur NE PEUT PAS supprimer une vente — uniquement annuler
   320→  isCancelled     Boolean  @default(false)
   321→  /// Motif d'annulation OBLIGATOIRE (tracé dans AuditLog)
   322→  cancellationReason String?
   323→
   324→  // --- OFFLINE-FIRST ---
   325→  /// Cette commande a été créée en mode hors ligne
   326→  isOfflineCreated Boolean @default(false)
   327→  /// Date à laquelle la commande a été synchronisée au serveur
   328→  syncedAt        DateTime?
   329→
   330→  businessId      String
   331→  business        Business @relation(fields: [businessId], references: [id])
   332→
   333→  customerId      String
   334→  customer        Customer @relation(fields: [customerId], references: [id])
   335→
   336→  /// ID de la promo appliquée à cette commande (optionnel)
   337→  promoId         String?
   338→
   339→  orderItems      OrderItem[]
   340→  payments        Payment[]
   341→  invoices        Invoice[]
   342→  commissions     Commission[]
   343→  /// Dettes liées à cette commande (si la dette vient d'une vente à crédit)
   344→  debts           Debt[]
   345→
   346→  /// userId qui a créé la commande
   347→  createdBy       String?
   348→  /// userId qui a annulé la commande (si applicable)
   349→  cancelledBy     String?
   350→
   351→  createdAt       DateTime @default(now())
   352→  updatedAt       DateTime @updatedAt
   353→}

   445→/// Session de caisse — ouverture et clôture avec calcul des écarts.
   446→/// Le montant théorique est calculé automatiquement à partir des ventes.
   447→/// Le montant réel est saisi par le gérant/patron.
   448→/// L'écart = montant réel - montant théorique (si ≠ 0, investigation nécessaire)
   449→model CashSession {
   450→  id              String   @id @default(cuid())
   451→  /// Date/heure d'ouverture de la caisse
   452→  openedAt        DateTime
   453→  /// Date/heure de clôture (null = caisse encore ouverte)
   454→  closedAt        DateTime?
   455→
   456→  /// Montant théorique = somme des paiements reçus pendant la session
   457→  theoreticalTotal Int?
   458→  /// Montant réel déclaré par le gérant/patron à la clôture
   459→  declaredTotal    Int?
   460→  /// Écart = declaredTotal - theoreticalTotal (calculé automatiquement)
   461→  difference       Int?
   462→
   463→  /// userId qui a ouvert la caisse
   464→  openedBy        String
   465→  /// userId qui a fermé la caisse
   466→  closedBy        String?
   467→
   468→  businessId      String
   469→  business        Business @relation(fields: [businessId], references: [id])
   470→
   471→  /// Caisse actuellement ouverte ?
   472→  isActive        Boolean  @default(true)
   473→
   474→  createdAt       DateTime @default(now())
   475→  updatedAt       DateTime @updatedAt
   476→}
