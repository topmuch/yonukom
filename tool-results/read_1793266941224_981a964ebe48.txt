   243→interface AppShellProps {
   244→  /** Currently active tab */

   252→export function AppShell({ activeTab, onTabChange, children }: AppShellProps) {
   253→  const config = useConfigStore((s) => s.config);
   254→  const user = useAuthStore((s) => s.user);
   255→  const [searchQuery, setSearchQuery] = useState("");
   256→  const [menuSheetOpen, setMenuSheetOpen] = useState(false);
   257→  const { logoUrl } = useBusinessLogo(user?.businessId);
   258→  const businessName = user?.businessName ?? "YonuKom";
   259→  const businessInitial = businessName.trim().charAt(0).toUpperCase() || "Y";
   260→
   261→  // --- Badge de notifications : compteur de commandes en ligne en attente ---
   262→  // Polling toutes les 30s pour détecter les nouvelles commandes PENDING.
   263→  // Affiché sur l'onglet "Cmds Online" (mobile bottom nav + desktop sidebar).
   264→  const onlineBusinessId = user?.businessId;
   265→  const [pendingOnlineCount, setPendingOnlineCount] = useState(0);
   266→  const fetchPendingOnlineCount = useCallback(async () => {
   267→    if (!onlineBusinessId) return;
   268→    try {
   269→      const res = await fetch(
   270→        `/api/online-orders?businessId=${onlineBusinessId}&status=PENDING`
   271→      );
   272→      if (!res.ok) return;
   273→      const data = await res.json();
   274→      setPendingOnlineCount(Array.isArray(data.orders) ? data.orders.length : 0);
   275→    } catch {
   276→      // Silencieux
   277→    }
   278→  }, [onlineBusinessId]);
   279→
   280→  useEffect(() => {
   281→    if (!onlineBusinessId) return;
   282→    fetchPendingOnlineCount();
   283→    const interval = setInterval(fetchPendingOnlineCount, 30000);
   284→    return () => clearInterval(interval);
   285→  }, [onlineBusinessId, fetchPendingOnlineCount]);
   286→
   287→  // Mode plein écran pour le POS — masque le sidebar et occupe toute la largeur
   288→  const isFullscreen = activeTab === "pos";
   289→
   290→  // Dynamic tabs based on active modules
   291→  const tabs = useMemo(() => {
   292→    const isB2B = config?.pipeline_b2b ?? false;
   293→    const isEcom = config?.ecommerce_social ?? false;
   294→    if (isB2B && !isEcom) {
   295→      return ALL_TABS.filter((t) => t.id !== "orders");
   296→    }
   297→    if (isB2B && isEcom) {
   298→      return ALL_TABS;
   299→    }
   300→    return ALL_TABS.filter((t) => t.id !== "pipeline");
   301→  }, [config?.pipeline_b2b, config?.ecommerce_social]);
   302→
   303→  const activeTabConfig =
   304→    tabs.find((t) => t.id === activeTab) ?? ALL_TABS[0];
   305→  const activeTabLabel = activeTabConfig.label;
   306→
   307→  return (
   308→    <div className="min-h-screen flex flex-col bg-background">
   309→      {/* ---- Sticky Header (mobile + tablet) ---- */}
   310→      <header className="lg:hidden sticky top-0 z-40 bg-background/85 backdrop-blur-xl border-b border-primary/10">
   311→        <div className="max-w-4xl mx-auto px-4 py-3 flex items-center justify-between">
   312→          <div className="flex items-center gap-2.5">
   313→            {logoUrl ? (
   314→              <img
   315→                src={logoUrl}
   316→                alt={businessName}
   317→                className="h-9 w-9 rounded-xl object-contain bg-white border border-slate-200 shrink-0"
   318→              />
   319→            ) : (
   320→              <div className="h-9 w-9 rounded-xl bg-gradient-to-br from-[#338BE2] to-[#2074C6] flex items-center justify-center shadow-md shrink-0">
   321→                <span className="text-white font-bold text-sm">
   322→                  {businessInitial}
   323→                </span>
   324→              </div>
   325→            )}
   326→            <div className="flex flex-col">
   327→              <span className="font-bold text-sm leading-tight text-foreground">
   328→                YonuKom
   329→              </span>
   330→              {user && (
   331→                <span className="text-[10px] text-muted-foreground leading-tight truncate max-w-[140px]">
   332→                  {user.businessName}
   333→                </span>
   334→              )}
   335→            </div>
   336→          </div>
   337→          <div className="flex items-center gap-1.5">
   338→            <RoleBadge />
   339→            <ConnectivityDot />
   340→            <ThemeToggle />
   341→          </div>
   342→        </div>
   343→      </header>
   344→
   345→      {/* ---- Floating menu button (visible only in fullscreen POS mode on desktop) ---- */}
   346→      {isFullscreen && (
   347→        <button
   348→          type="button"
   349→          onClick={() => onTabChange("home")}
   350→          className="hidden lg:flex fixed top-4 left-4 z-50 items-center gap-2 px-3 py-2 rounded-xl bg-background/90 backdrop-blur-md border border-border shadow-md text-sm font-medium hover:bg-accent transition-colors"
   351→          aria-label="Retour au menu"
   352→          title="Retour au menu principal"
   353→        >
   354→          <Menu className="w-4 h-4" />
   355→          <span className="hidden xl:inline">Menu</span>
   356→        </button>
   357→      )}
   358→
   359→      {/* ---- Content Area ---- */}
   360→      <main className={`flex-1 mx-auto w-full pb-20 lg:pb-6 ${isFullscreen ? "max-w-none lg:ml-0" : "max-w-7xl lg:ml-60"}`}>
   361→        {/* ---- Desktop Header (sticky top bar with search) ---- */}
   362→        {/* Masqué en mode plein écran (POS) car le POS a sa propre barre de recherche */}
   363→        <header className={`hidden sticky top-0 z-30 px-6 py-3.5 items-center justify-between bg-background/85 backdrop-blur-xl border-b border-primary/10 ${isFullscreen ? "lg:hidden" : "lg:flex"}`}>
   364→          <div className="flex items-center gap-3 flex-1 max-w-md">

   369→                value={searchQuery}
   370→                onChange={(e) => setSearchQuery(e.target.value)}
   371→                className="pl-10 pr-9 h-10 bg-muted border-primary/15 rounded-xl text-sm focus:bg-card focus:border-primary/30 transition-colors"
   372→              />
   373→              {searchQuery && (
   374→                <button
   375→                  onClick={() => setSearchQuery("")}
   376→                  className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   377→                  aria-label="Effacer"
   378→                >
   379→                  <X className="w-4 h-4" />
   380→                </button>
   381→              )}
   382→            </div>
   383→          </div>
   384→
   385→          <div className="flex items-center gap-2">
   386→            <button
   387→              className="w-9 h-9 rounded-lg bg-muted border border-primary/15 hover:bg-primary/5 hover:border-primary/30 flex items-center justify-center text-muted-foreground hover:text-primary transition-colors relative"
   388→              aria-label="Notifications"
   389→            >
   390→              <Bell className="w-4 h-4" />
   391→              <span className="absolute top-1.5 right-1.5 w-2 h-2 rounded-full bg-amber-400 ring-2 ring-background" />
   392→            </button>
   393→            <RoleBadge />
   394→            <ConnectivityDot />
   395→            <ThemeToggle />
   396→          </div>
   397→        </header>
   398→
   399→        <div className="px-4 lg:px-6 pt-4 lg:pt-6">{children}</div>
   400→      </main>
   401→
   402→      {/* ---- Bottom Navigation (Mobile) ---- */}
   403→      <nav
   404→        className="lg:hidden fixed bottom-0 inset-x-0 z-40 bg-background/95 backdrop-blur-xl border-t border-primary/10 safe-area-bottom"
   405→        role="tablist"
   406→        aria-label="Navigation principale"
   407→      >
   408→        <div className="max-w-4xl mx-auto flex items-center justify-around">
   409→          {tabs.map((tab) => {
   410→            const isActive = activeTab === tab.id;
   411→            const Icon = tab.icon;
   412→
   413→            return (
   414→              <button
   415→                key={tab.id}
   416→                role="tab"
   417→                aria-selected={isActive}
   418→                aria-label={tab.label}
   419→                onClick={() => onTabChange(tab.id)}
   420→                className={`
   421→                  flex flex-col items-center justify-center gap-1
   422→                  min-h-[56px] min-w-[48px] flex-1 py-2
   423→                  transition-colors duration-150 relative
   424→                  ${isActive ? "text-foreground" : "text-muted-foreground hover:text-foreground"}
   425→                `}
   426→                style={isActive ? { color: tab.color } : undefined}
   427→              >
   428→                {isActive ? (
   429→                  <motion.div
   430→                    layoutId="nav-icon"
   431→                    className="relative flex flex-col items-center"
   432→                  >
   433→                    <Icon className="w-5 h-5" strokeWidth={2.4} />
   434→                    <motion.div
   435→                      layoutId="nav-indicator"
   436→                      className={`absolute -bottom-1 h-0.5 w-5 rounded-full ${tab.accent}`}
   437→                      transition={{ type: "spring", stiffness: 350, damping: 30 }}
   438→                    />
   439→                  </motion.div>
   440→                ) : (
   441→                  <Icon className="w-5 h-5" strokeWidth={1.8} />
   442→                )}
   443→                {/* Badge notifications commandes en ligne */}
   444→                {tab.id === "online-orders" && pendingOnlineCount > 0 && (
   445→                  <span className="absolute top-1 right-1 min-w-[16px] h-4 px-1 rounded-full bg-amber-500 text-white text-[9px] font-bold flex items-center justify-center ring-2 ring-background animate-pulse">
   446→                    {pendingOnlineCount > 9 ? "9+" : pendingOnlineCount}
   447→                  </span>
   448→                )}
   449→                <span className="text-[10px] font-medium leading-none">
   450→                  {tab.label}
   451→                </span>
   452→              </button>
   453→            );
   454→          })}
   455→        </div>
   456→      </nav>
   457→
   458→      {/* ---- Desktop Sidebar (lg+) — BLUE with vibrant per-tab accents ---- */}
   459→      <aside className={`hidden fixed left-0 top-0 bottom-0 w-60 bg-gradient-to-b from-[#338BE2] to-[#2074C6] text-white shadow-[8px_0_24px_-12px_rgba(51,139,226,0.4)] flex-col z-40 ${isFullscreen ? "lg:hidden" : "lg:flex"}`}>
   460→        {/* Logo header */}
   461→        <div className="px-5 py-4 flex items-center gap-2.5 border-b border-white/15">
   462→          {logoUrl ? (
   463→            <img
   464→              src={logoUrl}
   465→              alt={businessName}
   466→              className="h-10 w-10 rounded-xl object-contain bg-white border border-white/40 shrink-0 shadow-md"
   467→            />
   468→          ) : (
   469→            <div className="w-10 h-10 rounded-xl bg-white flex items-center justify-center shadow-md shrink-0">
   470→              <span className="text-[#338BE2] font-bold text-base">
   471→                {businessInitial}
   472→              </span>
   473→            </div>
   474→          )}
   475→          <div className="flex flex-col min-w-0">
   476→            <span className="font-bold text-sm leading-tight text-white">
   477→              YonuKom
   478→            </span>
   479→            {user && (
   480→              <span className="text-[10px] text-white/70 leading-tight truncate max-w-[150px]">
   481→                {user.businessName}
   482→              </span>
   483→            )}
   484→          </div>
   485→        </div>
   486→
   487→        {/* Search mini */}
   488→        <div className="px-4 pt-4 pb-2">
   489→          <div className="relative">
   490→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-white/60" />
   491→            <input
   492→              type="text"
   493→              placeholder="Rechercher..."
   494→              className="w-full h-9 pl-9 pr-3 text-xs rounded-lg bg-white/10 border border-white/15 placeholder:text-white/50 text-white focus:outline-none focus:bg-white/20 focus:border-white/30 transition-colors"
   495→            />
   496→          </div>
   497→        </div>
   498→
   499→        {/* Nav — 3 sections (Principal / Gestion / Avancé) avec scroll visible fin.
   500→             Le sidebar contient 17 onglets — on les regroupe pour faciliter
   501→             la navigation, et on autorise un scroll discret si nécessaire. */}
   502→        <nav
   503→          className="flex-1 py-2 px-3 overflow-y-auto
   504→            [&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:bg-white/20 [&::-webkit-scrollbar-thumb]:rounded-full
   505→            [scrollbar-width:thin] [scrollbar-color:rgba(255,255,255,0.2)_transparent]"
   506→          role="tablist"
   507→          aria-label="Navigation principale"
   508→        >
   509→          <p className="px-3 pt-1 pb-1 text-[10px] font-semibold text-white/50 uppercase tracking-wider">
   510→            Principal
   511→          </p>
   512→          {tabs.filter(t => ["home", "pos", "products", "categories", "stock", "customers", "caisse", "orders", "online-orders"].includes(t.id)).map((tab) => {
   513→            const isActive = activeTab === tab.id;
   514→            const Icon = tab.icon;
   515→            return (
   516→              <button
   517→                key={tab.id}
   518→                role="tab"
   519→                aria-selected={isActive}
   520→                aria-label={tab.label}
   521→                onClick={() => onTabChange(tab.id)}
   522→                className={`
   523→                  w-full flex items-center gap-2.5 px-3 py-1.5 rounded-lg
   524→                  transition-all duration-150 min-h-[36px] font-medium text-[13px] relative overflow-hidden mb-0.5
   525→                  ${isActive ? "bg-white text-slate-900 shadow-md" : "text-white/90 hover:bg-white/15 hover:text-white"}
   526→                `}
   527→              >
   528→                {isActive && (
   529→                  <span className={`absolute left-0 top-0 bottom-0 w-1 bg-gradient-to-b ${tab.gradient}`} />
   530→                )}
   531→                <Icon
   532→                  className="w-4 h-4 relative z-10 shrink-0"
   533→                  strokeWidth={isActive ? 2.4 : 1.8}
   534→                  style={isActive ? { color: tab.color } : undefined}
   535→                />
   536→                <span className="relative z-10 truncate">{tab.label}</span>
   537→                {/* Badge notifications commandes en ligne (sidebar desktop) */}
   538→                {tab.id === "online-orders" && pendingOnlineCount > 0 && (
   539→                  <span className="ml-auto min-w-[18px] h-[18px] px-1 rounded-full bg-amber-500 text-white text-[10px] font-bold flex items-center justify-center shrink-0">
