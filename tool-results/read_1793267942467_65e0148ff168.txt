   504→// STEPPER NUMÉROTÉ PREMIUM (1-5) avec ligne connective
   505→// Completed = checkmark emerald, Current = blue filled, Upcoming = gray outline
   506→// ============================================================
   507→function NumberedStepper({ currentStepIndex }: { currentStepIndex: number }) {
   508→  const progressWidth = useMemo(
   509→    () =>
   510→      currentStepIndex === 0
   511→        ? 0
   512→        : (currentStepIndex / (ONBOARDING_STEPS.length - 1)) * 100,
   513→    [currentStepIndex],
   514→  );
   515→
   516→  return (
   517→    <div className="max-w-xl mx-auto w-full">
   518→      <div className="relative flex items-center justify-between">
   519→        {/* Ligne de fond */}
   520→        <div
   521→          className="absolute left-0 right-0 top-1/2 -translate-y-1/2 h-[2px] bg-border"
   522→          aria-hidden
   523→        />
   524→
   525→        {/* Ligne de progression (emerald) */}
   526→        <motion.div
   527→          className="absolute left-0 top-1/2 -translate-y-1/2 h-[2px] rounded-full"
   528→          style={{ backgroundColor: "#10B981" }}
   529→          initial={false}
   530→          animate={{ width: `${progressWidth}%` }}
   531→          transition={{ duration: 0.5, ease: "easeInOut" }}
   532→          aria-hidden
   533→        />
   534→
   535→        {/* Cercles numérotés */}
   536→        {ONBOARDING_STEPS.map((stepKey, i) => {
   537→          const isCurrent = i === currentStepIndex;
   538→          const isPast = i < currentStepIndex;
   539→
   540→          const bg = isPast
   541→            ? "#10B981"
   542→            : isCurrent
   543→            ? "#338BE2"
   544→            : "#FFFFFF";
   545→          const color = isPast || isCurrent ? "#FFFFFF" : "#94A3B8";
   546→          const border =
   547→            isPast || isCurrent ? "none" : "2px solid #E2E8F0";
   548→
   549→          return (
   550→            <div
   551→              key={stepKey}
   552→              className="relative z-10 flex flex-col items-center gap-2"
   553→            >
   554→              <motion.div
   555→                initial={false}
   556→                animate={{ scale: isCurrent ? 1.12 : 1 }}

   741→// ============================================================
   742→// CONFETTIS — animation CSS pure via framer-motion
   743→// ============================================================
   744→function Confetti() {
   745→  // Couleurs de la palette YonuKom
   746→  const colors = [
   747→    "#338BE2",
   748→    "#10B981",
   749→    "#B7BEDA",
   750→    "#FBBF24",
   751→    "#F472B6",
   752→    "#FFFFFF",
   753→  ];
   754→
   755→  // Génère 48 confettis avec positions / délais aléatoires stables
   756→  const pieces = useMemo(
   757→    () =>
   758→      Array.from({ length: 48 }, (_, i) => ({
   759→        id: i,
   760→        left: Math.random() * 100,
   761→        delay: Math.random() * 0.6,
   762→        duration: 2.6 + Math.random() * 2,
   763→        color: colors[i % colors.length],
   764→        size: 6 + Math.random() * 6,
   765→        rotate: Math.random() * 360,
   766→        drift: (Math.random() - 0.5) * 100,
   767→        shape: Math.random() > 0.5 ? "rect" : "circle",
   768→      })),
   769→    // eslint-disable-next-line react-hooks/exhaustive-deps
   770→    [],
   771→  );
   772→
   773→  return (
   774→    <div
   775→      className="absolute inset-0 pointer-events-none overflow-hidden"
   776→      aria-hidden
   777→    >
   778→      {pieces.map((p) => (
   779→        <motion.div
   780→          key={p.id}
   781→          initial={{ y: -30, x: 0, rotate: 0, opacity: 1 }}
   782→          animate={{
   783→            y: "110vh",
   784→            x: p.drift,
   785→            rotate: p.rotate + 720,
   786→            opacity: [1, 1, 0.8, 0],
   787→          }}
   788→          transition={{
   789→            duration: p.duration,
   790→            delay: p.delay,
   791→            repeat: Infinity,
   792→            ease: "easeIn",
   793→          }}
   794→          style={{
   795→            position: "absolute",
   796→            left: `${p.left}%`,
   797→            width: `${p.size}px`,
   798→            height: p.shape === "rect" ? `${p.size * 0.4}px` : `${p.size}px`,
   799→            backgroundColor: p.color,
   800→            borderRadius: p.shape === "rect" ? "2px" : "9999px",
   801→          }}
   802→        />
   803→      ))}
   804→    </div>
   805→  );
   806→}
