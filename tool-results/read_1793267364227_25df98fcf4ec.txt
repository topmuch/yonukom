    51→// --- Helpers ---
    52→function formatPrice(n: number) { return n.toLocaleString("fr-FR"); }
    53→function getOrCreateSessionId(): string {
    54→  if (typeof window === "undefined") return "ssr";
    55→  try {
    56→    let sid = localStorage.getItem("yonukom_session");
    57→    if (!sid) {
    58→      sid = "s-" + Math.random().toString(36).slice(2) + Date.now().toString(36);
    59→      localStorage.setItem("yonukom_session", sid);
    60→    }
    61→    return sid;
    62→  } catch { return "anon"; }
    63→}
