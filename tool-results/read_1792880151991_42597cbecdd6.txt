    66→export function SuperadminDashboard() {
    67→  const activePage = useSuperadminStore((s) => s.activePage);
    68→  const setActivePage = useSuperadminStore((s) => s.setActivePage);
    69→  const impersonation = useSuperadminStore((s) => s.impersonation);
    70→  const endImpersonation = useSuperadminStore((s) => s.endImpersonation);
    71→  const logout = useAuthStore((s) => s.logout);
    72→  const [sidebarOpen, setSidebarOpen] = useState(false);
    73→
    74→  // Fetch initial data
    75→  useEffect(() => {
    76→    fetchMetrics();
    77→    fetchTenants();
    78→  }, []);
    79→
    80→  const fetchMetrics = useCallback(async () => {
    81→    useSuperadminStore.getState().setMetricsLoading(true);
    82→    try {
    83→      const res = await fetch("/api/superadmin/analytics");
    84→      if (res.ok) {
    85→        const data = await res.json();
    86→        useSuperadminStore.getState().setMetrics(data);
    87→      }
    88→    } catch (err) {
    89→      console.error("Failed to fetch metrics:", err);
    90→    } finally {
    91→      useSuperadminStore.getState().setMetricsLoading(false);
    92→    }
    93→  }, []);
    94→
    95→  const fetchTenants = useCallback(async () => {
    96→    useSuperadminStore.getState().setTenantsLoading(true);
    97→    try {
    98→      const res = await fetch("/api/superadmin/tenants");
    99→      if (res.ok) {
   100→        const data = await res.json();
   101→        useSuperadminStore.getState().setTenants(data);
   102→      }
   103→    } catch (err) {
   104→      console.error("Failed to fetch tenants:", err);
   105→    } finally {
   106→      useSuperadminStore.getState().setTenantsLoading(false);
   107→    }
   108→  }, []);
   109→
   110→  const handleViewTenant = useCallback((id: string) => {
   111→    useSuperadminStore.getState().setSelectedTenantId(id);
   112→  }, []);
   113→
   114→  const handleImpersonate = useCallback(async (tenantId: string, tenantName: string) => {
   115→    try {
   116→      const res = await fetch("/api/superadmin/impersonate", {
   117→        method: "POST",
   118→        headers: { "Content-Type": "application/json" },
   119→        body: JSON.stringify({ tenantId }),
   120→      });
   121→      if (res.ok) {
   122→        useSuperadminStore.getState().startImpersonation(tenantId, tenantName);
   123→      }
   124→    } catch (err) {
   125→      console.error("Failed to start impersonation:", err);
   126→    }
   127→  }, []);
   128→
   129→  const selectedTenantId = useSuperadminStore((s) => s.selectedTenantId);
   130→  const tenants = useSuperadminStore((s) => s.tenants);
   131→  const selectedTenant = tenants.find((t) => t.id === selectedTenantId) ?? null;
   132→
   133→  return (
   134→    <div className="min-h-screen flex flex-col bg-background">
   135→      {/* Impersonation Banner */}
   136→      {impersonation.isImpersonating && impersonation.tenantName && (
   137→        <ImpersonationBanner
   138→          tenantName={impersonation.tenantName}
   139→          onEndImpersonation={endImpersonation}
   140→        />
   141→      )}
   142→
   143→      <div className="flex flex-1">
   144→        {/* Sidebar - Desktop */}
   145→        <aside className="hidden lg:flex w-64 flex-col border-r bg-card">
   146→          <SidebarContent
   147→            activePage={activePage}
   148→            onNavigate={(page) => { setActivePage(page); setSidebarOpen(false); }}
   149→            onLogout={logout}
   150→          />
   151→        </aside>
   152→
   153→        {/* Sidebar - Mobile (overlay) */}
   154→        <AnimatePresence>
   155→          {sidebarOpen && (
   156→            <>
   157→              <motion.div
   158→                initial={{ opacity: 0 }}
   159→                animate={{ opacity: 1 }}
   160→                exit={{ opacity: 0 }}
   161→                className="fixed inset-0 bg-black/40 z-40 lg:hidden"
   162→                onClick={() => setSidebarOpen(false)}
   163→              />
   164→              <motion.aside
   165→                initial={{ x: -280 }}
   166→                animate={{ x: 0 }}
   167→                exit={{ x: -280 }}
   168→                transition={{ type: "spring", stiffness: 300, damping: 30 }}
   169→                className="fixed left-0 top-0 bottom-0 w-[280px] z-50 bg-card border-r lg:hidden"
   170→              >
   171→                <SidebarContent
   172→                  activePage={activePage}
   173→                  onNavigate={(page) => { setActivePage(page); setSidebarOpen(false); }}
   174→                  onLogout={logout}
   175→                />
   176→              </motion.aside>
   177→            </>
   178→          )}
   179→        </AnimatePresence>
   180→
   181→        {/* Main Content */}
   182→        <main className="flex-1 min-w-0">
   183→          {/* Top bar (mobile) */}
   184→          <header className="lg:hidden flex items-center justify-between px-4 py-3 border-b bg-card">
   185→            <Button variant="ghost" size="sm" onClick={() => setSidebarOpen(true)}>
   186→              <BarChart3 className="w-5 h-5" />
   187→            </Button>
   188→            <div className="flex items-center gap-2">
   189→              <div className="w-8 h-8 rounded-lg bg-primary flex items-center justify-center">
   190→                <span className="text-primary-foreground font-bold text-xs">YK</span>
   191→              </div>
   192→              <span className="font-semibold text-sm">Superadmin</span>
   193→            </div>
   194→            <Badge variant="outline" className="text-xs border-yk-emerald/50 text-yk-emerald">
   195→              <Shield className="w-3 h-3 mr-1" />
   196→              Admin
   197→            </Badge>
   198→          </header>
   199→
   200→          {/* Page Content */}
   201→          <div className="p-4 lg:p-8">
   202→            <AnimatePresence mode="wait">
   203→              <motion.div
   204→                key={activePage}
   205→                initial={{ opacity: 0, y: 8 }}
   206→                animate={{ opacity: 1, y: 0 }}
   207→                exit={{ opacity: 0, y: -8 }}
   208→                transition={{ duration: 0.2 }}
   209→              >
   210→                {activePage === "overview" && (
   211→                  <OverviewPage
   212→                    onViewTenant={handleViewTenant}
   213→                    onImpersonate={handleImpersonate}
   214→                    onRefresh={fetchMetrics}
   215→                  />
   216→                )}
   217→                {activePage === "tenants" && (
   218→                  <TenantsPage
   219→                    onViewTenant={handleViewTenant}
   220→                    onImpersonate={handleImpersonate}
   221→                    onRefresh={fetchTenants}
   222→                  />
   223→                )}
   224→                {activePage === "subscriptions" && <SubscriptionsPage />}
   225→                {activePage === "analytics" && <AnalyticsPage onRefresh={fetchMetrics} />}
   226→                {activePage === "logs" && <LogsPage />}
   227→                {activePage === "settings" && <SettingsPage />}
   228→              </motion.div>
   229→            </AnimatePresence>
   230→          </div>
   231→        </main>
   232→      </div>
   233→
   234→      {/* Tenant Detail Panel */}
   235→      <TenantDetailPanel
   236→        tenant={selectedTenant}
   237→        open={!!selectedTenantId}
   238→        onOpenChange={(open) => {
   239→          if (!open) useSuperadminStore.getState().setSelectedTenantId(null);
   240→        }}
   241→        onImpersonate={handleImpersonate}
   242→      />
   243→    </div>
   244→  );
   245→}

   402→function TenantsPage({
   403→  onViewTenant,
   404→  onImpersonate,
   405→  onRefresh,
   406→}: {
   407→  onViewTenant: (id: string) => void;
   408→  onImpersonate: (tenantId: string, tenantName: string) => void;
   409→  onRefresh: () => void;
   410→}) {
   411→  const tenants = useSuperadminStore((s) => s.tenants);
   412→  const tenantsLoading = useSuperadminStore((s) => s.tenantsLoading);
   413→  const tenantSearch = useSuperadminStore((s) => s.tenantSearch);
   414→  const setTenantSearch = useSuperadminStore((s) => s.setTenantSearch);
   415→
   416→  const filteredTenants = tenants.filter((t) => {
   417→    if (!tenantSearch) return true;
   418→    const q = tenantSearch.toLowerCase();
   419→    return (
   420→      t.name.toLowerCase().includes(q) ||
   421→      t.ownerName.toLowerCase().includes(q) ||
   422→      t.ownerEmail.toLowerCase().includes(q)
   423→    );
   424→  });
   425→
   426→  return (
   427→    <div className="space-y-6">
   428→      {/* Header */}
   429→      <div className="flex items-center justify-between flex-wrap gap-3">
   430→        <div>
   431→          <h1 className="text-2xl font-bold">Tenants</h1>
   432→          <p className="text-muted-foreground text-sm">{tenants.length} commerces inscrits</p>
   433→        </div>
   434→        <div className="flex items-center gap-2">
   435→          <Button variant="outline" size="sm" onClick={onRefresh} disabled={tenantsLoading}>
   436→            <RefreshCw className={`w-4 h-4 mr-1.5 ${tenantsLoading ? "animate-spin" : ""}`} />
   437→            Actualiser
   438→          </Button>
   439→          <Button size="sm">
   440→            <Plus className="w-4 h-4 mr-1.5" />
   441→            Nouveau tenant
   442→          </Button>
   443→        </div>
   444→      </div>
   445→
   446→      {/* Search */}
   447→      <div className="relative max-w-md">
   448→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   449→        <Input
   450→          placeholder="Rechercher par nom, propriétaire ou email..."
   451→          value={tenantSearch}
   452→          onChange={(e) => setTenantSearch(e.target.value)}
   453→          className="pl-9"
   454→        />
   455→      </div>
   456→
   457→      {/* Table */}
   458→      <TenantTable
   459→        tenants={filteredTenants}
   460→        onViewTenant={onViewTenant}
   461→        onImpersonate={onImpersonate}
   462→      />
   463→    </div>
   464→  );
   465→}

   471→function SubscriptionsPage() {
   472→  const [subs, setSubs] = useState<Array<{
   473→    id: string;
   474→    plan: string;
   475→    status: string;
   476→    amount: number;
   477→    paymentMethod?: string;
   478→    nextBillingDate?: string;
   479→    tenant: { id: string; name: string; ownerName: string; ownerEmail: string };
   480→  }>>([]);
   481→  const [loading, setLoading] = useState(true);
   482→  const [statusFilter, setStatusFilter] = useState("all");
   483→
   484→  useEffect(() => {
   485→    (async () => {
   486→      setLoading(true);
   487→      try {
   488→        const url = statusFilter !== "all"
   489→          ? `/api/superadmin/subscriptions?status=${statusFilter}`
   490→          : "/api/superadmin/subscriptions";
   491→        const res = await fetch(url);
   492→        if (res.ok) setSubs(await res.json());
   493→      } catch (err) {
   494→        console.error("Failed to fetch subscriptions:", err);
   495→      } finally {
   496→        setLoading(false);
   497→      }
   498→    })();
   499→  }, [statusFilter]);
   500→
   501→  return (
   502→    <div className="space-y-6">
   503→      {/* Header */}
   504→      <div className="flex items-center justify-between flex-wrap gap-3">
   505→        <div>
   506→          <h1 className="text-2xl font-bold">Abonnements</h1>
   507→          <p className="text-muted-foreground text-sm">Gestion des abonnements tenants</p>
   508→        </div>
   509→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   510→          <SelectTrigger className="w-[180px]">
   511→            <SelectValue placeholder="Filtrer par statut" />
   512→          </SelectTrigger>
   513→          <SelectContent>
   514→            <SelectItem value="all">Tous</SelectItem>
   515→            <SelectItem value="active">🟢 Actifs</SelectItem>
   516→            <SelectItem value="past_due">🟠 En retard</SelectItem>
   517→            <SelectItem value="expired">🔴 Expirés</SelectItem>
   518→            <SelectItem value="cancelled">⚫ Annulés</SelectItem>
   519→          </SelectContent>
   520→        </Select>
   521→      </div>
   522→
   523→      {/* Subscription Cards Grid */}
   524→      {loading ? (
   525→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   526→          {Array.from({ length: 6 }).map((_, i) => (
   527→            <Card key={i} className="animate-pulse">
   528→              <CardContent className="p-6 space-y-3">
   529→                <div className="h-4 bg-muted rounded w-1/2" />
   530→                <div className="h-3 bg-muted rounded w-3/4" />
   531→                <div className="h-3 bg-muted rounded w-1/3" />
   532→              </CardContent>
   533→            </Card>
   534→          ))}
   535→        </div>
   536→      ) : subs.length === 0 ? (
   537→        <div className="text-center py-12">
   538→          <p className="text-muted-foreground">Aucun abonnement trouvé</p>
   539→        </div>
   540→      ) : (
   541→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   542→          {subs.map((sub, i) => (
   543→            <motion.div
   544→              key={sub.id}
   545→              initial={{ opacity: 0, y: 10 }}
   546→              animate={{ opacity: 1, y: 0 }}
   547→              transition={{ delay: i * 0.05 }}
   548→            >
   549→              <Card className="hover:shadow-md transition-shadow">
   550→                <CardContent className="p-5 space-y-3">
   551→                  <div className="flex items-start justify-between">
   552→                    <div className="min-w-0">
   553→                      <p className="font-semibold text-sm truncate">{sub.tenant.name}</p>
   554→                      <p className="text-xs text-muted-foreground truncate">{sub.tenant.ownerEmail}</p>
   555→                    </div>
   556→                    <SubscriptionStatusBadge status={sub.status as SubscriptionStatus} />
   557→                  </div>
   558→                  <Separator />
   559→                  <div className="flex items-center justify-between">
   560→                    <Badge variant="secondary" className="text-xs">
   561→                      {SUBSCRIPTION_PLAN_LABELS[sub.plan as SubscriptionPlan] ?? sub.plan}
   562→                    </Badge>

   683→function LogsPage() {
   684→  const platformLogs = useSuperadminStore((s) => s.platformLogs);
   685→  const setPlatformLogs = useSuperadminStore((s) => s.setPlatformLogs);
   686→  const [loading, setLoading] = useState(true);
   687→  const [levelFilter, setLevelFilter] = useState("all");
   688→
   689→  useEffect(() => {
   690→    (async () => {
   691→      setLoading(true);
   692→      try {
   693→        const url = levelFilter !== "all"
   694→          ? `/api/superadmin/logs?level=${levelFilter}`
   695→          : "/api/superadmin/logs";
   696→        const res = await fetch(url);
   697→        if (res.ok) setPlatformLogs(await res.json());
   698→      } catch (err) {
   699→        console.error("Failed to fetch logs:", err);
   700→      } finally {
   701→        setLoading(false);
   702→      }
   703→    })();
   704→  }, [levelFilter, setPlatformLogs]);
   705→
   706→  const levelColors: Record<string, string> = {
   707→    info: "text-blue-500",
   708→    warning: "text-amber-500",
   709→    error: "text-red-500",
   710→    critical: "text-red-700 font-bold",
   711→  };
   712→
   713→  const levelBg: Record<string, string> = {
   714→    info: "bg-blue-50",
   715→    warning: "bg-amber-50",
   716→    error: "bg-red-50",
   717→    critical: "bg-red-100",
   718→  };
   719→
   720→  return (
   721→    <div className="space-y-6">
   722→      {/* Header */}
   723→      <div className="flex items-center justify-between flex-wrap gap-3">
   724→        <div>
   725→          <h1 className="text-2xl font-bold">Logs plateforme</h1>
   726→          <p className="text-muted-foreground text-sm">Journal des événements globaux</p>
   727→        </div>
   728→        <Select value={levelFilter} onValueChange={setLevelFilter}>
   729→          <SelectTrigger className="w-[180px]">
   730→            <SelectValue placeholder="Niveau" />
   731→          </SelectTrigger>
   732→          <SelectContent>
   733→            <SelectItem value="all">Tous</SelectItem>
   734→            <SelectItem value="info">Info</SelectItem>
   735→            <SelectItem value="warning">Warning</SelectItem>
   736→            <SelectItem value="error">Error</SelectItem>
   737→            <SelectItem value="critical">Critical</SelectItem>
   738→          </SelectContent>
   739→        </Select>
   740→      </div>
   741→
   742→      {/* Logs list */}
   743→      {loading ? (
   744→        <div className="space-y-2">
   745→          {Array.from({ length: 5 }).map((_, i) => (
   746→            <div key={i} className="h-12 bg-muted rounded animate-pulse" />
   747→          ))}
   748→        </div>
   749→      ) : platformLogs.length === 0 ? (
   750→        <div className="text-center py-12">
   751→          <ScrollText className="w-12 h-12 text-muted-foreground mx-auto mb-3" />
   752→          <p className="text-muted-foreground">Aucun log trouvé</p>
   753→        </div>
   754→      ) : (
   755→        <div className="space-y-2 max-h-[600px] overflow-y-auto custom-scrollbar">
   756→          {platformLogs.map((log, i) => (
   757→            <motion.div
   758→              key={log.id}
   759→              initial={{ opacity: 0 }}
   760→              animate={{ opacity: 1 }}
   761→              transition={{ delay: i * 0.02 }}
   762→              className={`flex items-start gap-3 p-3 rounded-lg border ${levelBg[log.level] ?? "bg-card"}`}
   763→            >
   764→              <Badge variant="outline" className={`text-[9px] shrink-0 ${levelColors[log.level] ?? ""}`}>
   765→                {log.level.toUpperCase()}
   766→              </Badge>
   767→              <div className="flex-1 min-w-0">
   768→                <div className="flex items-center gap-2 mb-0.5">
   769→                  <Badge variant="secondary" className="text-[9px]">{log.category}</Badge>
   770→                  <span className="text-xs text-muted-foreground">
   771→                    {new Date(log.createdAt).toLocaleString("fr-FR")}
   772→                  </span>
   773→                </div>
   774→                <p className="text-sm truncate">{log.message}</p>
   775→              </div>
   776→            </motion.div>
   777→          ))}
   778→        </div>
   779→      )}
   780→    </div>
   781→  );
   782→}

   854→function MetricCard({
   855→  icon: Icon,
   856→  title,
   857→  value,
   858→  subtitle,
   859→  color,

   861→  icon: typeof CircleDollarSign;

   864→  subtitle: string;

   869→    amber: { icon: "text-amber-600", bg: "bg-amber-50" },
   870→  };
   871→  const c = colors[color];
   872→
   873→  return (
   874→    <Card>
   875→      <CardContent className="p-4">
   876→        <div className="flex items-start justify-between mb-2">
   877→          <p className="text-xs text-muted-foreground font-medium">{title}</p>
   878→          <div className={`p-1.5 rounded-md ${c.bg}`}>
   879→            <Icon className={`w-3.5 h-3.5 ${c.icon}`} />
   880→          </div>
   881→        </div>
   882→        <p className="text-2xl font-bold">{value}</p>
   883→        <p className="text-xs text-muted-foreground mt-1">{subtitle}</p>
