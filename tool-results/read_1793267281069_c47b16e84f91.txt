   131→export function SuperadminDashboard() {
   132→  const activePage = useSuperadminStore((s) => s.activePage);
   133→  const setActivePage = useSuperadminStore((s) => s.setActivePage);
   134→  const impersonation = useSuperadminStore((s) => s.impersonation);
   135→  const endImpersonation = useSuperadminStore((s) => s.endImpersonation);
   136→  const tenantSearch = useSuperadminStore((s) => s.tenantSearch);
   137→  const setTenantSearch = useSuperadminStore((s) => s.setTenantSearch);
   138→  const logout = useAuthStore((s) => s.logout);
   139→  const [sidebarOpen, setSidebarOpen] = useState(false);
   140→  const [notifCount, setNotifCount] = useState(0);
   141→
   142→  // Fetch initial data
   143→  const fetchMetrics = useCallback(async () => {
   144→    useSuperadminStore.getState().setMetricsLoading(true);
   145→    try {
   146→      const res = await fetch("/api/superadmin/analytics");
   147→      if (res.ok) {
   148→        const data = await res.json();
   149→        useSuperadminStore.getState().setMetrics(data);
   150→      }
   151→    } catch (err) {
   152→      console.error("Failed to fetch metrics:", err);
   153→    } finally {
   154→      useSuperadminStore.getState().setMetricsLoading(false);
   155→    }
   156→  }, []);
   157→
   158→  const fetchTenants = useCallback(async () => {
   159→    useSuperadminStore.getState().setTenantsLoading(true);
   160→    try {
   161→      const res = await fetch("/api/superadmin/tenants");
   162→      if (res.ok) {
   163→        const data = await res.json();
   164→        useSuperadminStore.getState().setTenants(data);
   165→      }
   166→    } catch (err) {
   167→      console.error("Failed to fetch tenants:", err);
   168→    } finally {
   169→      useSuperadminStore.getState().setTenantsLoading(false);
   170→    }
   171→  }, []);
   172→
   173→  // Fetch notification count (new tenants in last 24h + new messages + new users in last 24h)
   174→  const fetchNotifCount = useCallback(async () => {
   175→    try {
   176→      const since = new Date(Date.now() - 24 * 60 * 60 * 1000).toISOString();
   177→      const [tenantsRes, messagesRes, usersRes] = await Promise.all([
   178→        fetch(`/api/superadmin/tenants?sortBy=createdAt&since=${encodeURIComponent(since)}`),
   179→        fetch("/api/superadmin/messages?status=new"),
   180→        fetch("/api/superadmin/users"),
   181→      ]);
   182→      let count = 0;
   183→      if (tenantsRes.ok) {
   184→        const tenants = await tenantsRes.json();
   185→        count += Array.isArray(tenants)
   186→          ? tenants.filter((t: { createdAt?: string }) =>
   187→              t.createdAt ? new Date(t.createdAt) > new Date(since) : false
   188→            ).length
   189→          : 0;
   190→      }
   191→      if (messagesRes.ok) {
   192→        const data = await messagesRes.json();
   193→        count += data?.counts?.new ?? 0;
   194→      }
   195→      if (usersRes.ok) {
   196→        const users = await usersRes.json();
   197→        count += Array.isArray(users)
   198→          ? users.filter((u: { createdAt?: string }) =>
   199→              u.createdAt ? new Date(u.createdAt) > new Date(since) : false
   200→            ).length
   201→          : 0;
   202→      }
   203→      setNotifCount(count);
   204→    } catch (err) {
   205→      console.error("Failed to fetch notification count:", err);
   206→    }
   207→  }, []);
   208→
   209→  useEffect(() => {
   210→    fetchMetrics();
   211→    fetchTenants();
   212→    fetchNotifCount();
   213→  }, [fetchMetrics, fetchTenants, fetchNotifCount]);
   214→
   215→  const handleViewTenant = useCallback((id: string) => {
   216→    useSuperadminStore.getState().setSelectedTenantId(id);
   217→  }, []);
   218→
   219→  const handleImpersonate = useCallback(
   220→    async (tenantId: string, tenantName: string) => {
   221→      try {
   222→        const res = await fetch("/api/superadmin/impersonate", {
   223→          method: "POST",
   224→          headers: { "Content-Type": "application/json" },
   225→          body: JSON.stringify({ tenantId }),
   226→        });
   227→        if (res.ok) {
   228→          useSuperadminStore.getState().startImpersonation(tenantId, tenantName);
   229→        }
   230→      } catch (err) {
   231→        console.error("Failed to start impersonation:", err);
   232→      }
   233→    },
   234→    [],
   235→  );
   236→
   237→  const selectedTenantId = useSuperadminStore((s) => s.selectedTenantId);
   238→  const tenants = useSuperadminStore((s) => s.tenants);
   239→  const selectedTenant = tenants.find((t) => t.id === selectedTenantId) ?? null;
   240→
   241→  return (
   242→    <div className="min-h-screen flex flex-col bg-[#F6F9FE] dark:bg-background">
   243→      {/* Impersonation Banner */}
   244→      {impersonation.isImpersonating && impersonation.tenantName && (
   245→        <ImpersonationBanner
   246→          tenantName={impersonation.tenantName}
   247→          onEndImpersonation={endImpersonation}
   248→        />
   249→      )}
   250→
   251→      <div className="flex flex-1">
   252→        {/* Sidebar - Desktop (BLUE gradient, full height) */}
   253→        <aside className="hidden lg:flex w-64 flex-col fixed left-0 top-0 bottom-0 bg-gradient-to-b from-[#338BE2] to-[#2074C6] text-white shadow-[8px_0_24px_-12px_rgba(51,139,226,0.4)] z-40">
   254→          <SidebarContent
   255→            activePage={activePage}
   256→            onNavigate={(page) => {
   257→              setActivePage(page);
   258→              setSidebarOpen(false);
   259→            }}
   260→            onLogout={logout}
   261→            notifCount={notifCount}
   262→          />
   263→        </aside>
   264→
   265→        {/* Sidebar - Mobile (overlay) */}
   266→        <AnimatePresence>
   267→          {sidebarOpen && (
   268→            <>
   269→              <motion.div
   270→                initial={{ opacity: 0 }}
   271→                animate={{ opacity: 1 }}
   272→                exit={{ opacity: 0 }}
   273→                className="fixed inset-0 bg-black/50 backdrop-blur-sm z-40 lg:hidden"
   274→                onClick={() => setSidebarOpen(false)}
   275→              />
   276→              <motion.aside
   277→                initial={{ x: -300 }}
   278→                animate={{ x: 0 }}
   279→                exit={{ x: -300 }}
   280→                transition={{ type: "spring", stiffness: 300, damping: 30 }}
   281→                className="fixed left-0 top-0 bottom-0 w-[280px] z-50 bg-gradient-to-b from-[#338BE2] to-[#2074C6] text-white shadow-2xl lg:hidden"
   282→              >
   283→                <SidebarContent
   284→                  activePage={activePage}
   285→                  onNavigate={(page) => {
   286→                    setActivePage(page);
   287→                    setSidebarOpen(false);
   288→                  }}
   289→                  onLogout={logout}
   290→                  notifCount={notifCount}
   291→                />
   292→              </motion.aside>
   293→            </>
   294→          )}
   295→        </AnimatePresence>
   296→
   297→        {/* Main Content — pushed right by sidebar */}
   298→        <main className="flex-1 min-w-0 flex flex-col lg:ml-64">
   299→          {/* Top bar — BLUE with search */}
   300→          <header className="hidden lg:flex sticky top-0 z-30 bg-gradient-to-r from-[#338BE2] to-[#2074C6] text-white items-center justify-between px-6 py-3 shadow-md">
   301→            {/* Left: page title */}
   302→            <div className="flex items-center gap-3">
   303→              <div className="w-9 h-9 rounded-lg bg-white/15 backdrop-blur flex items-center justify-center">
   304→                {(() => {
   305→                  const Item = NAV_ITEMS.find((n) => n.page === activePage)?.icon ?? BarChart3;
   306→                  return <Item className="w-4.5 h-4.5 text-white" />;
   307→                })()}
   308→              </div>
   309→              <div>
   310→                <p className="text-xs text-white/70 font-medium uppercase tracking-wider">
   311→                  Superadmin
   312→                </p>
   313→                <p className="text-sm font-bold text-white leading-tight">
   314→                  {NAV_ITEMS.find((n) => n.page === activePage)?.label ?? "Vue d'ensemble"}
   315→                </p>
   316→              </div>
   317→            </div>
   318→
   319→            {/* Center: search bar */}
   320→            <div className="flex-1 max-w-md mx-8">
   321→              <div className="relative">
   322→                <Search className="absolute left-3.5 top-1/2 -translate-y-1/2 w-4 h-4 text-white/60" />
   323→                <input
   324→                  type="text"
   325→                  placeholder="Rechercher un tenant, email..."
   326→                  value={tenantSearch}
   327→                  onChange={(e) => setTenantSearch(e.target.value)}
   328→                  className="w-full h-10 pl-10 pr-9 rounded-xl bg-white/15 backdrop-blur border border-white/20 placeholder:text-white/60 text-white text-sm focus:outline-none focus:bg-white/25 focus:border-white/40 transition-colors"
   329→                />
   330→                {tenantSearch && (
   331→                  <button
   332→                    onClick={() => setTenantSearch("")}
   333→                    className="absolute right-3 top-1/2 -translate-y-1/2 text-white/60 hover:text-white"
   334→                    aria-label="Effacer"
   335→                  >
   336→                    <X className="w-4 h-4" />
   337→                  </button>
   338→                )}
   339→              </div>
   340→            </div>
   341→
   342→            {/* Right: admin badge + actions */}
   343→            <div className="flex items-center gap-2">
   344→              <button
   345→                className="w-9 h-9 rounded-lg bg-white/15 backdrop-blur hover:bg-white/25 flex items-center justify-center relative transition-colors"
   346→                aria-label="Notifications"
   347→                onClick={() => setActivePage("overview")}
   348→              >
   349→                <Bell className="w-4 h-4 text-white" />
   350→                {notifCount > 0 && (
   351→                  <span className="absolute -top-1 -right-1 min-w-[18px] h-[18px] px-1 rounded-full bg-amber-300 text-[#2074C6] text-[10px] font-bold flex items-center justify-center ring-2 ring-[#338BE2]">
   352→                    {notifCount > 99 ? "99+" : notifCount}
   353→                  </span>
   354→                )}
   355→              </button>
   356→              <Badge className="bg-white/15 text-white border border-white/20 backdrop-blur">
   357→                <Shield className="w-3 h-3 mr-1" />
   358→                Admin
   359→              </Badge>
   360→              <div className="[&_button]:text-white [&_button:hover]:bg-white/15">
   361→                <ThemeToggle />
   362→              </div>
   363→            </div>
   364→          </header>
   365→
   366→          {/* Top bar (mobile) */}
   367→          <header className="lg:hidden sticky top-0 z-30 bg-gradient-to-r from-[#338BE2] to-[#2074C6] text-white flex items-center justify-between px-4 py-3 shadow-md">
   368→            <button
   369→              onClick={() => setSidebarOpen(true)}
   370→              aria-label="Ouvrir le menu"
   371→              className="w-9 h-9 rounded-lg bg-white/15 flex items-center justify-center"
   372→            >
   373→              <BarChart3 className="w-5 h-5" />
   374→            </button>
   375→            <div className="flex items-center gap-2">
   376→              <div className="w-8 h-8 rounded-lg bg-white flex items-center justify-center">
   377→                <span className="text-[#338BE2] font-bold text-xs">YK</span>
   378→              </div>
   379→              <span className="font-semibold text-sm">Superadmin</span>
   380→            </div>
   381→            <div className="flex items-center gap-1">
   382→              <div className="[&_button]:text-white [&_button:hover]:bg-white/15">
   383→                <ThemeToggle />
   384→              </div>
   385→              <Badge className="text-xs bg-white/15 text-white border border-white/20">
   386→                <Shield className="w-3 h-3 mr-1" />
   387→                Admin
   388→              </Badge>
   389→            </div>
   390→          </header>
   391→
   392→          {/* Page Content */}
   393→          <div className="flex-1 p-4 lg:p-6">
   394→            <AnimatePresence mode="wait">
   395→              <motion.div
   396→                key={activePage}
   397→                initial={{ opacity: 0, y: 8 }}
   398→                animate={{ opacity: 1, y: 0 }}
   399→                exit={{ opacity: 0, y: -8 }}
   400→                transition={{ duration: 0.2 }}
   401→              >
   402→                {activePage === "overview" && (
   403→                  <OverviewPage
   404→                    onViewTenant={handleViewTenant}
   405→                    onImpersonate={handleImpersonate}
   406→                    onRefresh={fetchMetrics}
   407→                  />
   408→                )}
   409→                {activePage === "tenants" && (
   410→                  <TenantsPage
   411→                    onViewTenant={handleViewTenant}
   412→                    onImpersonate={handleImpersonate}
   413→                    onRefresh={fetchTenants}
   414→                  />
   415→                )}
   416→                {activePage === "users" && <UsersPage />}
   417→                {activePage === "subscriptions" && <SubscriptionsPage />}
   418→                {activePage === "analytics" && <AnalyticsPage onRefresh={fetchMetrics} />}
   419→                {activePage === "messages" && <MessagesPage />}
   420→                {activePage === "logs" && <LogsPage />}
   421→                {activePage === "settings" && <SettingsPage />}
   422→              </motion.div>
   423→            </AnimatePresence>
   424→          </div>
   425→        </main>
   426→      </div>
   427→
   428→      {/* Tenant Detail Panel */}
   429→      <TenantDetailPanel
   430→        tenant={selectedTenant}
   431→        open={!!selectedTenantId}
   432→        onOpenChange={(open) => {
   433→          if (!open) useSuperadminStore.getState().setSelectedTenantId(null);
   434→        }}
   435→        onImpersonate={handleImpersonate}
   436→        onRefresh={fetchTenants}
   437→      />
   438→    </div>
   439→  );
   440→}

   513→        })}

   516→      <Separator className="my-2 bg-white/15" />
   517→
   518→      {/* Admin badge + logout */}
   519→      <div className="p-3 space-y-2">
   520→        <div className="flex items-center gap-2.5 px-3 py-2 rounded-xl bg-white/10">
   521→          <div className="w-8 h-8 rounded-lg bg-white/15 flex items-center justify-center">
   522→            <Shield className="w-4 h-4 text-white" />
   523→          </div>
   524→          <div className="min-w-0 flex-1">
   525→            <p className="text-xs font-semibold text-white truncate">Superadmin</p>

   545→function CircularProgressRing({
   546→  value,
   547→  size = 70,
   548→  stroke = 6,
   549→  color = BLUE,
   550→  trackColor = "rgba(51,139,226,0.12)",
   551→  label,
   552→  sublabel,
   553→}: {
   554→  value: number;
   555→  size?: number;
   556→  stroke?: number;
   557→  color?: string;
   558→  trackColor?: string;
   559→  label?: string;
   560→  sublabel?: string;
   561→}) {
   562→  const radius = (size - stroke) / 2;
   563→  const circumference = 2 * Math.PI * radius;
   564→  const clamped = Math.max(0, Math.min(100, value));
   565→  const offset = circumference - (clamped / 100) * circumference;
   566→
   567→  return (
   568→    <div className="relative" style={{ width: size, height: size }}>
   569→      <svg width={size} height={size} className="-rotate-90">
   570→        <circle
   571→          cx={size / 2}
   572→          cy={size / 2}
   573→          r={radius}
   574→          fill="none"
   575→          stroke={trackColor}
   576→          strokeWidth={stroke}
   577→        />
   578→        <motion.circle
   579→          cx={size / 2}
   580→          cy={size / 2}
   581→          r={radius}
   582→          fill="none"
   583→          stroke={color}
   584→          strokeWidth={stroke}
   585→          strokeDasharray={circumference}
   586→          strokeLinecap="round"
   587→          initial={{ strokeDashoffset: circumference }}
   588→          animate={{ strokeDashoffset: offset }}
   589→          transition={{ duration: 1, ease: "easeOut" }}
   590→        />
   591→      </svg>
   592→      {label && (
   593→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   594→          <span
   595→            className="text-base font-bold leading-none"
   596→            style={{ color }}
   597→          >
   598→            {label}
   599→          </span>
   600→          {sublabel && (
   601→            <span className="text-[9px] text-muted-foreground mt-0.5">{sublabel}</span>
   602→          )}
   603→        </div>
   604→      )}
   605→    </div>
   606→  );

   667→function AreaChart({ height = 220 }: { height?: number }) {
   668→  const data = [40, 55, 45, 70, 65, 85, 75, 95, 88, 110, 105, 125];
   669→  const months = ["J", "F", "M", "A", "M", "J", "J", "A", "S", "O", "N", "D"];

   672→  const h = height;
   673→  const pad = 20;

   675→  const points = data.map((v, i) => {
   676→    const x = pad + i * stepX;
   677→    const y = h - pad - (v / max) * (h - pad * 2);

   682→    if (points.length === 0) return "";

   687→      const cx = (x0 + x1) / 2;
   688→      d += ` C ${cx} ${y0}, ${cx} ${y1}, ${x1} ${y1}`;
   689→    }
   690→    return d;
   691→  })();
   692→
   693→  const areaD = `${pathD} L ${points[points.length - 1][0]} ${h - pad} L ${points[0][0]} ${h - pad} Z`;

   709→          <stop offset="0%" stopColor={BLUE} />
   710→          <stop offset="100%" stopColor={BLUE_LIGHT} />
   711→        </linearGradient>
   712→      </defs>
   713→
   714→      {[0.25, 0.5, 0.75, 1].map((p) => (
   715→        <line
   716→          key={p}
   717→          x1={pad}
   718→          x2={w - pad}
   719→          y1={h - pad - p * (h - pad * 2)}
   720→          y2={h - pad - p * (h - pad * 2)}
   721→          stroke="currentColor"
   722→          strokeOpacity="0.15"
   723→          strokeWidth="1"
   724→          strokeDasharray="4 4"
   725→          className="text-muted-foreground"
   726→        />
   727→      ))}
   728→
   729→      <motion.path
   730→        d={areaD}
   731→        fill="url(#areaGrad)"
   732→        initial={{ opacity: 0 }}
   733→        animate={{ opacity: 1 }}
   734→        transition={{ duration: 0.8, delay: 0.3 }}
   735→      />
   736→
   737→      <motion.path
   738→        d={pathD}
   739→        fill="none"
   740→        stroke="url(#lineGrad)"
   741→        strokeWidth="2.5"
   742→        strokeLinecap="round"
   743→        initial={{ pathLength: 0 }}
   744→        animate={{ pathLength: 1 }}
   745→        transition={{ duration: 1.2, ease: "easeOut" }}
   746→      />
   747→
   748→      {points.map(([x, y], i) => (
   749→        <motion.circle
   750→          key={i}
   751→          cx={x}
   752→          cy={y}
   753→          r={3.5}
   754→          fill="white"
   755→          stroke={BLUE}
   756→          strokeWidth="2"
   757→          initial={{ scale: 0 }}
   758→          animate={{ scale: 1 }}
   759→          transition={{ delay: 0.5 + i * 0.05 }}
   760→        />
   761→      ))}
   762→
   763→      {months.map((m, i) => (
   764→        <text
   765→          key={i}
   766→          x={pad + i * stepX}
   767→          y={h - 5}
   768→          textAnchor="middle"
   769→          className="fill-muted-foreground"
   770→          style={{ fontSize: "10px" }}
   771→        >
   772→          {m}
   773→        </text>
   774→      ))}
   775→    </svg>
   776→  );

   779→function DonutChart({
   780→  segments,
   781→  size = 180,
   782→  stroke = 24,
   783→  centerLabel,
   784→  centerSub,
   785→  centerDelta,
   786→}: {
   787→  segments: { value: number; color: string; label: string }[];
   788→  size?: number;
   789→  stroke?: number;
   790→  centerLabel?: string;
   791→  centerSub?: string;
   792→  centerDelta?: string;
   793→}) {
   794→  const radius = (size - stroke) / 2;
   795→  const cx = size / 2;
   796→  const cy = size / 2;
   797→  const total = segments.reduce((sum, s) => sum + s.value, 0) || 1;
   798→  const circumference = 2 * Math.PI * radius;
   799→
   800→  // Compute cumulative offsets in a pure way (no outer mutation, lint-friendly).
   801→  // First pass: build cumulative fractions array. Second pass: derive arc data.
   802→  const fractions = segments.map((s) => s.value / total);
   803→  const cumulativeFractions: number[] = [];
   804→  for (let k = 0; k < fractions.length; k++) {
   805→    const prev = k > 0 ? cumulativeFractions[k - 1] : 0;
   806→    cumulativeFractions.push(prev + fractions[k]);
   807→  }
   808→  const arcs = segments.map((seg, i) => {
   809→    const fraction = fractions[i];
   810→    const dashLen = fraction * circumference;
   811→    const prevCumulative = i > 0 ? cumulativeFractions[i - 1] : 0;
   812→    const dashOffset = -prevCumulative * circumference;
   813→    return { ...seg, dashLen, dashOffset, i };
   814→  });
   815→
   816→  return (
   817→    <div className="relative" style={{ width: size, height: size }}>
   818→      <svg width={size} height={size} className="-rotate-90">
   819→        <circle
   820→          cx={cx}
   821→          cy={cy}
   822→          r={radius}
   823→          fill="none"
   824→          stroke="rgba(51,139,226,0.06)"
   825→          strokeWidth={stroke}
   826→        />
   827→        {arcs.map((arc) => (
   828→          <motion.circle
   829→            key={arc.i}
   830→            cx={cx}
   831→            cy={cy}
   832→            r={radius}
   833→            fill="none"
   834→            stroke={arc.color}
   835→            strokeWidth={stroke}
   836→            strokeDasharray={`${arc.dashLen} ${circumference - arc.dashLen}`}
   837→            initial={{ strokeDashoffset: 0 }}
   838→            animate={{ strokeDashoffset: arc.dashOffset }}
   839→            transition={{ duration: 1, delay: 0.2 + arc.i * 0.15, ease: "easeOut" }}
   840→          />
   841→        ))}
   842→      </svg>
   843→      {centerLabel && (
   844→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   845→          <span className="text-2xl font-bold text-foreground">{centerLabel}</span>
   846→          {centerSub && (
   847→            <span className="text-xs text-muted-foreground mt-0.5">{centerSub}</span>
   848→          )}
   849→          {centerDelta && (
   850→            <span className="text-xs font-semibold text-emerald-600 mt-0.5 flex items-center gap-0.5">
   851→              <ArrowUpRight className="w-3 h-3" />
   852→              {centerDelta}
   853→            </span>
   854→          )}
   855→        </div>
   856→      )}
   857→    </div>
   858→  );
   859→}
